# robots.txt — 검색 크롤러의 접근 허용/차단 규칙
# User-agent: * → 모든 크롤러(Googlebot, Bingbot 등)에 적용
# Allow: / → 기본적으로 전체 허용
# Disallow: 인증·관리자 경로는 인덱싱 제외 (개인정보·보안)

User-agent: *
Allow: /

# 관리자 페이지: 외부 노출 불필요, 보안상 크롤링 차단
Disallow: /admin

# OAuth 콜백 경로: 동적 토큰 포함, 인덱싱 의미 없음
Disallow: /auth

# 마이페이지: 로그인 필요 페이지, 크롤러에 유용한 정보 없음
Disallow: /mypage

# sitemap 위치 명시 → 크롤러가 자동으로 발견
Sitemap: https://coupang-severance-app.vercel.app/sitemap.xml
